## 5. Prisma schema draft

```prisma
model Account {
  id              String        @id @default(uuid())
  email           String        @unique
  fullName        String        @map("full_name")
  status          AccountStatus @default(pending_verification)
  emailVerifiedAt DateTime?     @map("email_verified_at")
  lastLoginAt     DateTime?     @map("last_login_at")
  createdAt       DateTime      @default(now()) @map("created_at")
  updatedAt       DateTime      @updatedAt @map("updated_at")

  credentials     AccountCredential[]
  tokens          AccountToken[]
  sessions        Session[]
  memberships     TenantMembership[]
  sentInvitations MembershipInvitation[] @relation("MembershipInvitationInviter")

  @@index([status])
  @@map("accounts")
}

model Organization {
  id         String             @id @default(uuid())
  name       String             @unique
  slug       String             @unique
  status     OrganizationStatus @default(active)
  verifiedAt DateTime?          @map("verified_at")
  createdAt  DateTime           @default(now()) @map("created_at")
  updatedAt  DateTime           @updatedAt @map("updated_at")
  deletedAt  DateTime?          @map("deleted_at")

  memberships TenantMembership[]
  invitations MembershipInvitation[]
  roles       Role[]
  auditLogs   AuditLog[]

  @@index([status])
  @@index([deletedAt])
  @@map("organizations")
}

model TenantMembership {
  id             String           @id @default(uuid())
  organizationId String           @map("organization_id")
  accountId      String           @map("account_id")
  status         MembershipStatus @default(invited)
  source         MembershipSource
  activatedAt    DateTime?        @map("activated_at")
  deactivatedAt  DateTime?        @map("deactivated_at")
  createdAt      DateTime         @default(now()) @map("created_at")
  updatedAt      DateTime         @updatedAt @map("updated_at")

  organization Organization @relation(fields: [organizationId], references: [id], onDelete: Restrict, onUpdate: Cascade)
  account      Account      @relation(fields: [accountId], references: [id], onDelete: Restrict, onUpdate: Cascade)

  @@unique([organizationId, accountId])
  @@index([accountId, status])
  @@index([organizationId, status])
  @@map("tenant_memberships")
}

model MembershipInvitation {
  id                 String                     @id @default(uuid())
  organizationId     String                     @map("organization_id")
  email              String
  tokenHash          String                     @unique @map("token_hash")
  status             MembershipInvitationStatus @default(pending)
  invitedByAccountId String                     @map("invited_by_account_id")
  expiresAt          DateTime                   @map("expires_at")
  acceptedAt         DateTime?                  @map("accepted_at")
  revokedAt          DateTime?                  @map("revoked_at")
  createdAt          DateTime                   @default(now()) @map("created_at")
  updatedAt          DateTime                   @updatedAt @map("updated_at")

  organization     Organization @relation(fields: [organizationId], references: [id], onDelete: Restrict, onUpdate: Cascade)
  invitedByAccount Account      @relation("MembershipInvitationInviter", fields: [invitedByAccountId], references: [id], onDelete: Restrict, onUpdate: Cascade)

  @@index([organizationId, status])
  @@index([email, status])
  @@index([expiresAt])
  @@map("membership_invitations")
}

model Role {
  id             String    @id @default(uuid())
  organizationId String?   @map("organization_id")
  key            String
  name           String
  scope          RoleScope @default(tenant)
  description    String?
  createdAt      DateTime  @default(now()) @map("created_at")
  updatedAt      DateTime  @updatedAt @map("updated_at")

  organization    Organization?    @relation(fields: [organizationId], references: [id], onDelete: SetNull, onUpdate: Cascade)
  rolePermissions RolePermission[]

  @@unique([organizationId, key, scope])
  @@map("roles")
}

model AuditLog {
  id             String      @id @default(uuid())
  organizationId String?     @map("organization_id")
  action         AuditAction
  resource       String?
  resourceId     String?     @map("resource_id")
  metadata       Json?
  requestId      String?     @map("request_id")
  createdAt      DateTime    @default(now()) @map("created_at")

  organization Organization? @relation(fields: [organizationId], references: [id], onDelete: SetNull, onUpdate: Cascade)

  @@index([organizationId, action])
  @@map("audit_logs")
}
```

ALTER TABLE "organizations" ALTER COLUMN "status" TYPE "OrganizationStatus" USING ("status"::text::"OrganizationStatus");
ALTER TABLE "organizations" ALTER COLUMN "status" SET DEFAULT 'active';

ALTER TABLE "roles" RENAME COLUMN "tenant_id" TO "organization_id";

ALTER TABLE "audit_logs" ADD CONSTRAINT "audit_logs_organization_id_fkey" FOREIGN KEY ("organization_id") REFERENCES "organizations"("id") ON DELETE SET NULL ON UPDATE CASCADE;

  "id" TEXT NOT NULL,

  "updated_at" TIMESTAMP(3) NOT NULL,

  "id" TEXT NOT NULL,
  "organization_id" TEXT NOT NULL,
  "email" TEXT NOT NULL,
  "token_hash" TEXT NOT NULL,

  "invited_by_account_id" TEXT NOT NULL,
  "expires_at" TIMESTAMP(3) NOT NULL,
  "accepted_at" TIMESTAMP(3),
  "revoked_at" TIMESTAMP(3),
  "created_at" TIMESTAMP(3) NOT NULL DEFAULT CURRENT_TIMESTAMP,

CREATE INDEX "tenant_memberships_organization_id_status_idx" ON "tenant_memberships"("organization_id", "status");

ALTER TABLE "tenant_memberships" ADD CONSTRAINT "tenant_memberships_organization_id_fkey" FOREIGN KEY ("organization_id") REFERENCES "organizations"("id") ON DELETE RESTRICT ON UPDATE CASCADE;
ALTER TABLE "tenant_memberships" ADD CONSTRAINT "tenant_memberships_account_id_fkey" FOREIGN KEY ("account_id") REFERENCES "accounts"("id") ON DELETE RESTRICT ON UPDATE CASCADE;
ALTER TABLE "membership_invitations" ADD CONSTRAINT "membership_invitations_organization_id_fkey" FOREIGN KEY ("organization_id") REFERENCES "organizations"("id") ON DELETE RESTRICT ON UPDATE CASCADE;

DROP TYPE IF EXISTS "TenantStatus";

## 7. Design rationale

- `Organization` thay `Tenant` để khớp thuật ngữ khóa luận; tenant vẫn là boundary logic, organization là root entity.
- `Account` không có `organizationId`; membership là quan hệ nhiều-nhiều có trạng thái.
- `verifiedAt` nullable vì organization có thể tồn tại trước khi được kích hoạt/xác minh.
- `deletedAt` giữ soft delete vì organization là tenant root, hard delete có rủi ro mất evidence.
- `TenantMembership.status` bảo vệ invariant inactive membership không tạo tenant context.
- `@@unique([organizationId, accountId])` chặn duplicate membership/race condition.
- `MembershipInvitation.tokenHash` là sensitive; raw token không được lưu DB/return API/log.
- Partial unique pending invitation theo `organization_id,email` chặn spam duplicate pending invite nhưng vẫn cho mời lại sau accepted/revoked/expired.
- FK membership/invitation dùng `Restrict` để tránh xóa account/organization làm mất lịch sử membership.
- `Role.organizationId` và `AuditLog.organizationId` giữ compatibility với RBAC/audit scaffold nhưng đổi tên khỏi legacy `tenant_id`.
